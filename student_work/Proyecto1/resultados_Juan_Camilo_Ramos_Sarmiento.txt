class DataProcessor:
    def __init__(self):
        self.queue = ArrayQueue()
        self.stack = ArrayStack()
        self.lista = []
        self.redo_stack = ArrayStack()   # para el bonus

    # ---------- 3. add ----------
    def add(self, record):
        if not isinstance(record, tuple):
            raise ValueError("El registro debe ser una tupla (sensor, variable, value).")
        if len(record) != 3:
            raise ValueError("El registro debe tener exactamente 3 elementos.")
        sensor, variable, value = record
        if not isinstance(value, (int, float)) or isinstance(value, bool):
            raise TypeError("El valor debe ser numérico (int o float).")
        self.queue.enqueue(record)

    # ---------- 4. process_next ----------
    def process_next(self):
        if self.queue.is_empty():
            raise Empty("No hay registros pendientes.")

        record = self.queue.dequeue()
        sensor, variable, value = record
        indice = self._buscar(sensor, variable)

        if indice is None:
            self.stack.push(("NUEVO", sensor, variable, None, value))
            self.lista.append(record)
        else:
            valor_viejo = self.lista[indice][2]
            self.stack.push(("ACTUALIZAR", sensor, variable, valor_viejo, value))
            self.lista[indice] = record

        while not self.redo_stack.is_empty():
            self.redo_stack.pop()

        return record

    # ---------- 5. undo ----------
    def undo(self):
        if self.stack.is_empty():
            raise Empty("No hay cambios para deshacer.")

        tipo, sensor, variable, valor_viejo, valor_nuevo = self.stack.pop()
        indice = self._buscar(sensor, variable)

        if tipo == "NUEVO":
            if indice is not None:
                self.lista.pop(indice)
        elif tipo == "ACTUALIZAR":
            if indice is not None:
                self.lista[indice] = (sensor, variable, valor_viejo)

        self.redo_stack.push((tipo, sensor, variable, valor_viejo, valor_nuevo))

    # ---------- 6. pending ----------
    def pending(self):
        return len(self.queue)

    # ---------- 7. current_value ----------
    def current_value(self, sensor, variable):
        indice = self._buscar(sensor, variable)
        if indice is None:
            raise KeyError(f"No existe registro para ({sensor}, {variable}).")
        return self.lista[indice][2]

    # ---------- 12. Bonus: redo ----------
    def redo(self):
        if self.redo_stack.is_empty():
            raise Empty("No hay cambios para rehacer.")

        tipo, sensor, variable, valor_viejo, valor_nuevo = self.redo_stack.pop()

        if tipo == "NUEVO":
            self.lista.append((sensor, variable, valor_nuevo))
        elif tipo == "ACTUALIZAR":
            indice = self._buscar(sensor, variable)
            if indice is not None:
                self.lista[indice] = (sensor, variable, valor_nuevo)

        self.stack.push((tipo, sensor, variable, valor_viejo, valor_nuevo))

    # ---------- auxiliar ----------
    def _buscar(self, sensor, variable):
        for i, reg in enumerate(self.lista):
            if reg[0] == sensor and reg[1] == variable:
                return i
        return None
